async function init() { try { const config = await loadConfig(); supabaseClient = window.supabase.createClient( config.supabaseUrl, config.supabaseAnonKey, { auth: { detectSessionInUrl: true, persistSession: true, autoRefreshToken: true } } ); // Supabase PKCE recovery mengirimkan ?code=... const url = new URL(window.location.href); const code = url.searchParams.get("code"); if (code) { const { error } = await supabaseClient.auth.exchangeCodeForSession(code); if (error) { throw error; } // Hapus code dari address bar setelah berhasil diproses window.history.replaceState( {}, document.title, window.location.pathname ); } // Beri waktu untuk event recovery diproses await new Promise(resolve => setTimeout(resolve, 500)); const { data: { session }, error } = await supabaseClient.auth.getSession(); if (error) { throw error; } if (!session) { throw new Error( "Tautan reset password tidak valid atau sudah kedaluwarsa." ); } loading.style.display = "none"; formArea.style.display = "block"; } catch (error) { loading.style.display = "none"; formArea.style.display = "block"; saveButton.disabled = true; showMessage( error?.message || "Tautan reset password tidak dapat digunakan.", "error" ); } }
